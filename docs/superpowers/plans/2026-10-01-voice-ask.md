# Voice Ask Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A global mic button that lets visitors ask KiwiCue in speech or text and streams matching events, movies and markets onto a full-screen overlay, with cost caps.

**Architecture:** A Next.js route (`POST /api/ask`) streams NDJSON events. It reserves quota in Supabase, asks TypeSafe Jev (typed judgements) and DeepSeek (open slots) in parallel, merges filters, loads candidates from the existing loaders, ranks them with Jev and streams a one-sentence DeepSeek reply. The client keeps conversation state and renders the stream.

**Tech Stack:** Next.js 16 route handlers, React 19, TypeScript, zod 4, Vercel AI SDK `ai@7.0.123` with `@ai-sdk/typesafe-ai@3.0.11` and `@ai-sdk/deepseek@3.0.57`, Supabase (Postgres RPC), Vitest + Testing Library, Playwright.

Spec: `docs/superpowers/specs/2026-10-01-voice-ask-design.md`.

## Global Constraints

- Secrets `TYPESAFE_AI_API_KEY`, `DEEPSEEK_API_KEY`, `ASK_HASH_SALT` are server-only; never `NEXT_PUBLIC_`, never logged, never committed.
- Jev model id: `jev-latest`. DeepSeek model id: `deepseek-flash`.
- Prices (USD per 1M tokens, peak, upper bound): Jev input 0.04, output free; DeepSeek input 0.30, output 1.20.
- Quotas per Auckland day: guest 5, free (signed in) 20, plus 200; burst 10 per minute for every tier.
- Budget: NZ$0.50 per Auckland day, NZ$8 per Auckland month, USD→NZD 1.75, degrade at 80 %.
- DeepSeek slot extraction timeout 2.5 s; reply timeout 6 s. Jev answers with confidence < 0.6 count as "unmentioned".
- Request text 1–300 characters; at most 5 earlier turns; at most 40 candidate cards.
- Design tests only allow radius tokens (`var(--radius-*)`) and ban `radial-gradient`. Use `--portal-*` colour tokens.
- Files in this repo use CRLF line endings; keep new files CRLF (run `node -e` normaliser shown in Task 1).
- Every task ends with its own commit. No `Co-Authored-By` lines.
- Functions under 40 lines; no `any`; no empty `catch`; no `console.log`.

```ts
export type AskFailureStep = "jev" | "deepseek" | "reply" | "quota" | "source" | "stream";

function field(error: unknown, key: string): unknown {
  return error && typeof error === "object" && key in error ? (error as Record<string, unknown>)[key] : undefined;
}

/** Records why an ask step failed. User text never reaches the log, only the error's shape. */
export function logAskFailure(step: AskFailureStep, error: unknown): void {
  const status = field(error, "statusCode") ?? field(error, "status");
  const code = field(error, "code");
  console.error(JSON.stringify({
    event: "ask_failure",
    step,
    status: typeof status === "number" ? status : undefined,
    code: typeof code === "string" ? code.slice(0, 60) : undefined,
    name: error instanceof Error ? error.name : undefined,
  }));
}
```

```ts
import { askStreamEventSchema, type AskStreamEvent } from "./schema";

const encoder = new TextEncoder();
const MAX_LINE_LENGTH = 16_384;

export function encodeEvent(event: AskStreamEvent): Uint8Array {
  return encoder.encode(`${JSON.stringify(event)}\n`);
}

function parseLine(line: string): AskStreamEvent | null {
  if (!line.trim() || line.length > MAX_LINE_LENGTH) return null;
  try {
    const parsed = askStreamEventSchema.safeParse(JSON.parse(line));
    return parsed.success ? parsed.data : null;
  } catch (error) {
    if (error instanceof SyntaxError) return null; // A broken line is skipped; the rest of the stream is still useful.
    throw error;
  }
}

/** Reads an NDJSON body into validated events, ignoring anything that does not match the schema. */
export async function* readEvents(body: ReadableStream<Uint8Array>): AsyncGenerator<AskStreamEvent> {
  const reader = body.pipeThrough(new TextDecoderStream()).getReader();
  let buffer = "";
  while (true) {
    const { value, done } = await reader.read();
    if (done) break;
    buffer += value;
    const lines = buffer.split("\n");
    buffer = lines.pop() ?? "";
    for (const line of lines) {
      const event = parseLine(line);
      if (event) yield event;
    }
  }
  const last = parseLine(buffer);
  if (last) yield last;
}
```

```ts
import { Temporal } from "@js-temporal/polyfill";
import { addDays, aucklandDate, datesForFilter } from "../cinemas/auckland-time";

export type AskWhen = "any" | "today" | "tomorrow" | "weekend" | "week";
const MAX_DAYS_AHEAD = 60;
const DEFAULT_RANGE_DAYS = 30;
const ISO_DATE = /^\d{4}-\d{2}-\d{2}$/u;

export function datesForWhen(when: AskWhen, now: Date): string[] {
  if (when === "any") return [];
  return datesForFilter(when === "week" ? "all" : when, now);
}

function isRealDate(value: string): boolean {
  if (!ISO_DATE.test(value)) return false;
  try {
    return Temporal.PlainDate.from(value, { overflow: "reject" }).toString() === value;
  } catch (error) {
    if (error instanceof RangeError) return false;
    throw error;
  }
}

export function sanitizeDates(dates: readonly string[], now: Date): string[] {
  const today = aucklandDate(now);
  const last = addDays(today, MAX_DAYS_AHEAD);
  return [...new Set(dates)].filter((date) => isRealDate(date) && date >= today && date <= last).sort().slice(0, 14);
}

function startOfAucklandDay(date: string): Date {
  const instant = Temporal.PlainDate.from(date).toZonedDateTime({ timeZone: "Pacific/Auckland" }).toInstant();
  return new Date(instant.epochMilliseconds);
}

/** The instant range covering the given Auckland dates, never starting in the past. */
export function dateRange(dates: readonly string[], now: Date): { start: Date; end: Date } {
  if (dates.length === 0) return { start: now, end: new Date(now.getTime() + DEFAULT_RANGE_DAYS * 86_400_000) };
  const first = startOfAucklandDay(dates[0]);
  const end = startOfAucklandDay(addDays(dates[dates.length - 1], 1));
  return { start: first.getTime() > now.getTime() ? first : now, end };
}
```

```ts
export function chipsFor(filters: AskFilters, language: Language): AskChip[] {
  const text = labels[language];
  const chips: AskChip[] = [];
  if (filters.domain !== "any") chips.push({ key: "domain", label: text.domain[filters.domain] });
  if (filters.dates.length > 0) chips.push({ key: "dates", label: dateLabel(filters.dates, language) });
  if (filters.timeOfDay !== "any") chips.push({ key: "timeOfDay", label: text.timeOfDay[filters.timeOfDay] });
  if (filters.area !== "any") chips.push({ key: "area", label: text.area[filters.area] });
  if (filters.category !== "any") chips.push({ key: "category", label: text.category[filters.category] });
  if (filters.kids) chips.push({ key: "kids", label: text.kids });
  if (filters.free) chips.push({ key: "free", label: text.free });
  if (filters.place) chips.push({ key: "place", label: filters.place });

  if (key.startsWith("name:")) return { ...filters, names: filters.names.filter((_, index) => `name:${index}` !== key) };

    case "domain": return { ...filters, domain: "any" };
    case "dates": return { ...filters, dates: [] };
    case "timeOfDay": return { ...filters, timeOfDay: "any" };
    case "area": return { ...filters, area: "any" };
    case "category": return { ...filters, category: "any" };
    case "kids": return { ...filters, kids: false };
    case "free": return { ...filters, free: false };
    case "place": return { ...filters, place: null };
    default: return filters;
  }
```

```ts
export type JevAnswer =
  | { type: "choice"; choice: string; probabilities?: Record<string, number> }
  | { type: "score"; score: number; probabilities?: Record<string, number> }

export type Evaluate = (input: {
  state: Record<string, JSONValue>;
  questions: Record<string, JevQuestion>;
  abortSignal?: AbortSignal;

export const jevEvaluate: Evaluate = async ({ state, questions, abortSignal }) => {
  const result = await experimental_evaluate({
    model: typeSafeAi.evaluationModel(JEV_MODEL), state, questions, abortSignal, maxRetries: 1,
  });
  return { answers: result.answers as Record<string, JevAnswer>, inputTokens: result.usage.inputTokens ?? 0 };
```

```ts
import type { AskArea, AskCategory, AskDomain, AskFilters, AskTime } from "./schema";

// Every question offers "unmentioned" so a follow-up like "make it evening" only changes the time.
const QUESTIONS = {
  domain: { type: "choice", instructions: "What kind of outing is the latest message asking about?", criteria: { ...UNMENTIONED, any: "Anything to do, no type preference", events: "Concerts, shows, sport, festivals or other ticketed events", movies: "Films at a cinema", markets: "Markets, night markets, farmers markets" } },

  kids: { type: "choice", instructions: "Does the latest message say whether children are coming?", criteria: { ...UNMENTIONED, on: "Children are coming or it must suit kids", off: "Explicitly no longer needs to suit kids" } },
  free: { type: "choice", instructions: "Does the latest message ask about free entry?", criteria: { ...UNMENTIONED, on: "Only free things", off: "Explicitly fine to pay" } },

function confident(answer: JevAnswer | undefined): string | undefined {
  if (!answer || answer.type !== "choice" || answer.choice === "unmentioned") return undefined;
  const probability = answer.probabilities?.[answer.choice] ?? 1;
  return probability >= MIN_CONFIDENCE ? answer.choice : undefined;
}

function toUnderstanding(answers: Record<string, JevAnswer>): JevUnderstanding {
  const pick = <T extends string>(key: keyof typeof QUESTIONS) => confident(answers[key]) as T | undefined;
  const entries = Object.entries({
    domain: pick<AskDomain>("domain"), when: pick<AskWhen>("when"), timeOfDay: pick<AskTime>("timeOfDay"),
    area: pick<AskArea>("area"), category: pick<AskCategory>("category"), kids: pick<Toggle>("kids"), free: pick<Toggle>("free"),
  }).filter(([, value]) => value !== undefined);
  return Object.fromEntries(entries) as JevUnderstanding;
}

export async function understandWithJev(input: {
  text: string; history: string[]; previous: AskFilters | null; today: string; evaluate?: Evaluate; abortSignal?: AbortSignal;
}): Promise<{ understanding: JevUnderstanding; inputTokens: number }> {
  const state = {
    latestMessage: input.text, earlierMessages: input.history, currentFilters: input.previous,
    today: input.today, city: "Auckland, New Zealand",
  };
  const { answers, inputTokens } = await (input.evaluate ?? jevEvaluate)({ state, questions: QUESTIONS, abortSignal: input.abortSignal });
  return { understanding: toUnderstanding(answers), inputTokens };
}
```

```ts
import { deepSeek } from "@ai-sdk/deepseek";
import { generateText, Output } from "ai";
import { z } from "zod";

import type { AskFilters } from "./schema";

export const DEEPSEEK_MODEL = "deepseek-flash";
const DEFAULT_TIMEOUT_MS = 2_500;

const slotsSchema = z.object({
  names: z.array(z.string().trim().min(1).max(80)).max(5),
  place: z.string().trim().min(1).max(80).nullable(),
  dates: z.array(z.string().regex(/^\d{4}-\d{2}-\d{2}$/u)).max(14),
  mode: z.enum(["refine", "new"]),
});
export type DeepSeekUnderstanding = z.infer<typeof slotsSchema>;
export type TokenUsage = { inputTokens: number; outputTokens: number };
export type GenerateSlots = (input: { system: string; prompt: string; abortSignal: AbortSignal }) => Promise<{ output: DeepSeekUnderstanding; usage: TokenUsage }>;

export const deepseekGenerateSlots: GenerateSlots = async ({ system, prompt, abortSignal }) => {
  const result = await generateText({
    model: deepSeek(DEEPSEEK_MODEL), system, prompt, abortSignal, maxRetries: 0,
    output: Output.object({ schema: slotsSchema }),
  });
  return { output: result.output, usage: { inputTokens: result.usage.inputTokens ?? 0, outputTokens: result.usage.outputTokens ?? 0 } };
};

function systemPrompt(today: string): string {
  return [
    "You extract search details from what a visitor to KiwiCue, an Auckland events guide, just said.",
    `Today is ${today} in Auckland, New Zealand.`,
    "names: specific artists, shows, films, teams or venues they named, as written. Empty if none.",
    "place: a suburb or venue they want to be near, or null.",
    "dates: exact calendar dates (YYYY-MM-DD) only when they named specific days such as \"Saturday\" or \"3 October\"; otherwise empty.",
    "mode: \"refine\" when the latest message adjusts the earlier request, \"new\" when it starts a different one.",
  ].join("\n");
}

/** Open-ended details Jev cannot enumerate. Returns null when DeepSeek is slow or fails. */
export async function understandWithDeepSeek(input: {
  text: string; history: string[]; previous: AskFilters | null; today: string; generate?: GenerateSlots; timeoutMs?: number;
}): Promise<{ understanding: DeepSeekUnderstanding; usage: TokenUsage } | null> {
  const prompt = JSON.stringify({ latestMessage: input.text, earlierMessages: input.history, currentFilters: input.previous });
  try {
    const result = await (input.generate ?? deepseekGenerateSlots)({
      system: systemPrompt(input.today), prompt, abortSignal: AbortSignal.timeout(input.timeoutMs ?? DEFAULT_TIMEOUT_MS),
    });
    const parsed = slotsSchema.safeParse(result.output);
    return parsed.success ? { understanding: parsed.data, usage: result.usage } : null;
  } catch (error) {
    logAskFailure("deepseek", error);
    return null;
  }
}
```

```ts
import { datesForWhen, sanitizeDates } from "./resolve-dates";
import { EMPTY_FILTERS, type AskFilters } from "./schema";
import type { DeepSeekUnderstanding } from "./understand-deepseek";
import type { JevUnderstanding, Toggle } from "./understand-jev";

function toggle(value: Toggle | undefined, current: boolean): boolean {
  return value === undefined ? current : value === "on";
}

function mergeDates(base: AskFilters, jev: JevUnderstanding | null, deepseek: DeepSeekUnderstanding | null, now: Date): string[] {
  const explicit = sanitizeDates(deepseek?.dates ?? [], now);
  if (explicit.length > 0) return explicit;
  return jev?.when === undefined ? base.dates : datesForWhen(jev.when, now);
}

/**
 * A follow-up only changes what it mentions: Jev answers "unmentioned" for everything else, so
 * those keys keep the previous value. A new question (DeepSeek mode "new") starts from scratch.
 */
export function mergeFilters(input: {
  previous: AskFilters | null; jev: JevUnderstanding | null; deepseek: DeepSeekUnderstanding | null; now: Date;
}): AskFilters {
  const { jev, deepseek } = input;
  const base = !input.previous || deepseek?.mode === "new" ? EMPTY_FILTERS : input.previous;
  return {
    domain: jev?.domain ?? base.domain,
    dates: mergeDates(base, jev, deepseek, input.now),
    timeOfDay: jev?.timeOfDay ?? base.timeOfDay,
    area: jev?.area ?? base.area,
    category: jev?.category ?? base.category,
    kids: toggle(jev?.kids, base.kids),
    free: toggle(jev?.free, base.free),
    names: deepseek && deepseek.names.length > 0 ? deepseek.names : base.names,
    place: deepseek?.place ?? base.place,
  };
}
```

```ts
export type AskUsage = { jevInputTokens: number; deepseekInputTokens: number; deepseekOutputTokens: number };
export const ZERO_USAGE: AskUsage = { jevInputTokens: 0, deepseekInputTokens: 0, deepseekOutputTokens: 0 };

// USD per 1M tokens at peak rates; per-token micro-USD is the same number. Checked 2026-10-01
// on docs.typesafe.ai and api-docs.deepseek.com. Using peak rates keeps the budget conservative.
const PRICE = { jevInput: 0.04, deepseekInput: 0.30, deepseekOutput: 1.20 } as const;

export function addUsage(total: AskUsage, extra: Partial<AskUsage>): AskUsage {
  return {
    jevInputTokens: total.jevInputTokens + (extra.jevInputTokens ?? 0),
    deepseekInputTokens: total.deepseekInputTokens + (extra.deepseekInputTokens ?? 0),
    deepseekOutputTokens: total.deepseekOutputTokens + (extra.deepseekOutputTokens ?? 0),
  };
}

export function costMicroUsd(usage: AskUsage): number {
  return Math.ceil(usage.jevInputTokens * PRICE.jevInput
    + usage.deepseekInputTokens * PRICE.deepseekInput
    + usage.deepseekOutputTokens * PRICE.deepseekOutput);
}
```

```ts
import "server-only";
import { createHash } from "node:crypto";
import { createClient } from "@supabase/supabase-js";
import { readSupabasePublicConfig } from "../supabase/config";
import { logAskFailure } from "./log";
import type { AskTier } from "./schema";

export type QuotaDecision =
  | { allowed: true; usageId: string; remaining: number; degrade: boolean; tier: AskTier }
  | { allowed: false; reason: "rate_limited" | "quota_exhausted" | "budget_exhausted" | "unavailable"; remaining: number; tier: AskTier };
export type QuotaStore = {
  reserve(subject: string, tier: AskTier, userId: string | null): Promise<QuotaDecision>;
  recordCost(usageId: string, costMicroUsd: number, degraded: boolean): Promise<void>;
};

type ReserveRow = { allowed: boolean; reason: string | null; usage_id: string | null; remaining: number; degrade: boolean };
const DENIALS = new Set(["rate_limited", "quota_exhausted", "budget_exhausted"]);

export function clientIp(headers: Headers): string {
  const forwarded = headers.get("x-forwarded-for")?.split(",")[0]?.trim();
  const candidate = forwarded || headers.get("x-real-ip")?.trim() || "unknown";
  return candidate.slice(0, 64);
}

/** Guests are counted by a salted hash so raw IP addresses are never stored. */
export function hashSubject(ip: string, salt: string): string {
  return createHash("sha256").update(`${salt}:${ip}`).digest("hex");
}

function toDecision(row: ReserveRow | undefined, tier: AskTier): QuotaDecision {
  if (row?.allowed && row.usage_id) return { allowed: true, usageId: row.usage_id, remaining: row.remaining, degrade: row.degrade, tier };
  const reason = row?.reason && DENIALS.has(row.reason) ? row.reason as "rate_limited" | "quota_exhausted" | "budget_exhausted" : "unavailable";
  return { allowed: false, reason, remaining: row?.remaining ?? 0, tier };
}

export function createSupabaseQuotaStore(env: Record<string, string | undefined> = process.env): QuotaStore | null {
  const config = readSupabasePublicConfig(env);
  const secret = env.SUPABASE_SECRET_KEY || env.SUPABASE_SERVICE_ROLE_KEY;
  if (!config || !secret) return null;
  const client = createClient(config.url, secret, { auth: { persistSession: false, autoRefreshToken: false } });
  return {
    async reserve(subject, tier, userId) {
      const { data, error } = await client.rpc("ask_reserve", { p_subject: subject, p_tier: tier, p_user: userId });
      if (error) { logAskFailure("quota", error); return toDecision(undefined, tier); }
      return toDecision((data as ReserveRow[] | null)?.[0], tier);
    },
    async recordCost(usageId, cost, degraded) {
      const { error } = await client.rpc("ask_record_cost", { p_usage_id: usageId, p_cost: cost, p_degraded: degraded });
      if (error) logAskFailure("quota", error);
    },
  };
}
```

```sql
-- Voice ask: per-person quotas and a global spend cap. Only the server (secret key) can read or
-- write these tables; RLS is on with no policies and the functions are service-role only.
create table public.ask_limits (
  tier text primary key check (tier in ('guest', 'free', 'plus')),
  per_day integer not null check (per_day >= 0),
  per_minute integer not null check (per_minute >= 0)
);
insert into public.ask_limits (tier, per_day, per_minute) values
  ('guest', 5, 10), ('free', 20, 10), ('plus', 200, 10);

create table public.ask_settings (
  id integer primary key default 1 check (id = 1),
  daily_budget_nzd_cents integer not null default 50 check (daily_budget_nzd_cents >= 0),
  monthly_budget_nzd_cents integer not null default 800 check (monthly_budget_nzd_cents >= 0),
  usd_to_nzd numeric not null default 1.75 check (usd_to_nzd > 0),
  degrade_ratio numeric not null default 0.8 check (degrade_ratio > 0 and degrade_ratio <= 1)
);
insert into public.ask_settings default values;

create table public.ask_usage (
  id uuid primary key default gen_random_uuid(),
  subject text not null check (length(subject) <= 80),
  tier text not null references public.ask_limits (tier),
  user_id uuid references auth.users (id) on delete cascade,
  created_at timestamptz not null default now(),
  cost_micro_usd integer not null default 0 check (cost_micro_usd >= 0),
  degraded boolean not null default false
);
create index ask_usage_subject_created on public.ask_usage (subject, created_at desc);
create index ask_usage_created on public.ask_usage (created_at);

alter table public.ask_limits enable row level security;
alter table public.ask_settings enable row level security;
alter table public.ask_usage enable row level security;
revoke all on public.ask_limits, public.ask_settings, public.ask_usage from anon, authenticated;

create function public.ask_reserve(p_subject text, p_tier text, p_user uuid)
returns table (allowed boolean, reason text, usage_id uuid, remaining integer, degrade boolean)
language plpgsql security definer set search_path = '' as $$
declare
  v_limits public.ask_limits%rowtype;
  v_settings public.ask_settings%rowtype;
  v_day_start timestamptz := date_trunc('day', now() at time zone 'Pacific/Auckland') at time zone 'Pacific/Auckland';
  v_month_start timestamptz := date_trunc('month', now() at time zone 'Pacific/Auckland') at time zone 'Pacific/Auckland';
  v_minute integer; v_today integer; v_day_spend bigint; v_month_spend bigint;
  v_day_cap numeric; v_month_cap numeric; v_degrade boolean; v_id uuid;
begin
  -- Serialise requests from one subject so two tabs cannot both take the last free ask.
  perform pg_advisory_xact_lock(hashtext(p_subject));
  select * into v_limits from public.ask_limits l where l.tier = p_tier;
  if not found then return query select false, 'invalid_tier', null::uuid, 0, false; return; end if;
  select * into v_settings from public.ask_settings s where s.id = 1;
  delete from public.ask_usage u where u.created_at < now() - interval '40 days';

  select count(*) filter (where u.created_at > now() - interval '1 minute'),
         count(*) filter (where u.created_at >= v_day_start)
    into v_minute, v_today
    from public.ask_usage u
   where u.subject = p_subject and u.created_at >= least(v_day_start, now() - interval '1 minute');
  if v_minute >= v_limits.per_minute then
    return query select false, 'rate_limited', null::uuid, greatest(v_limits.per_day - v_today, 0), false; return;
  end if;
  if v_today >= v_limits.per_day then
    return query select false, 'quota_exhausted', null::uuid, 0, false; return;
  end if;

  select coalesce(sum(u.cost_micro_usd) filter (where u.created_at >= v_day_start), 0), coalesce(sum(u.cost_micro_usd), 0)
    into v_day_spend, v_month_spend
    from public.ask_usage u where u.created_at >= v_month_start;
  -- NZD cents → micro-USD: cents / 100 / rate × 1,000,000.
  v_day_cap := v_settings.daily_budget_nzd_cents * 10000 / v_settings.usd_to_nzd;
  v_month_cap := v_settings.monthly_budget_nzd_cents * 10000 / v_settings.usd_to_nzd;
  if v_day_spend >= v_day_cap or v_month_spend >= v_month_cap then
    return query select false, 'budget_exhausted', null::uuid, greatest(v_limits.per_day - v_today, 0), false; return;
  end if;

  v_degrade := v_day_spend >= v_day_cap * v_settings.degrade_ratio or v_month_spend >= v_month_cap * v_settings.degrade_ratio;
  insert into public.ask_usage (subject, tier, user_id, degraded) values (p_subject, p_tier, p_user, v_degrade)
    returning id into v_id;
  return query select true, null::text, v_id, v_limits.per_day - v_today - 1, v_degrade;
end;
$$;

create function public.ask_record_cost(p_usage_id uuid, p_cost integer, p_degraded boolean)
returns void language sql security definer set search_path = '' as $$
  update public.ask_usage set cost_micro_usd = greatest(p_cost, 0), degraded = degraded or p_degraded
   where id = p_usage_id;
$$;

revoke execute on function public.ask_reserve(text, text, uuid) from public, anon, authenticated;
revoke execute on function public.ask_record_cost(uuid, integer, boolean) from public, anon, authenticated;
grant execute on function public.ask_reserve(text, text, uuid) to service_role;
grant execute on function public.ask_record_cost(uuid, integer, boolean) to service_role;
```

```ts
function wants(filters: AskFilters, domain: "events" | "movies" | "markets"): boolean {
  if (filters.domain !== "any") return filters.domain === domain;
  // Films have no category, price or family tag to filter on, except kids via rating.
  return domain !== "movies" || (filters.category === "any" && !filters.free);
}

async function settle<T>(source: "events" | "movies", load: () => Promise<T[]>, unavailable: CandidateResult["unavailable"]): Promise<T[]> {
  try {
    return await load();
  } catch (error) {
    logAskFailure("source", error);
    unavailable.push(source);
    return [];
  }
}

export async function loadCandidates(input: { filters: AskFilters; language: Language; now: Date; sources?: CandidateSources }): Promise<CandidateResult> {
  const { filters, language, now } = input;
  const sources = input.sources ?? defaultSources;
  const unavailable: CandidateResult["unavailable"] = [];
  const [events, screenings] = await Promise.all([
    wants(filters, "events") ? settle("events", () => sources.events(dateRange(filters.dates, now), filters.names[0] ?? null), unavailable) : [],
    wants(filters, "movies") ? settle("movies", () => sources.movies(now), unavailable) : [],
  ]);
  const markets = wants(filters, "markets") || filters.domain === "any" ? sources.markets(now) : [];
  const upcoming = [...events, ...markets].filter((event) => Date.parse(eventStart(event)) >= now.getTime() || !event.start.dateTime);
  const cards = [
    ...eventCards(applyPlace(upcoming.filter((event) => keepEvent(event, filters)), filters.place), language),
    ...movieCards(screenings.filter((screening) => Date.parse(screening.startTime) >= now.getTime() && keepScreening(screening, filters))),
  ];
  return { cards: cards.sort((left, right) => left.when.localeCompare(right.when)).slice(0, MAX_CANDIDATES), unavailable };
}
```

```ts
import "server-only";
import { normalizedMovieTitle } from "../verified-movie-sessions";
import { jevEvaluate, type Evaluate, type JevQuestion } from "./jev";
import { logAskFailure } from "./log";
import type { AskCard, AskFilters } from "./schema";

export const BEST_MIN_SCORE = 0.5;
const LEVELS = ["Unrelated to the request", "Loosely related", "A good fit", "Exactly what was asked for"];

function byDate(left: AskCard, right: AskCard): number {
  return left.when.localeCompare(right.when);
}

function isNamed(card: AskCard, names: readonly string[]): boolean {
  const title = normalizedMovieTitle(card.title);
  return names.some((name) => { const needle = normalizedMovieTitle(name); return needle.length > 0 && title.includes(needle); });
}

function question(card: AskCard): JevQuestion {
  return { type: "score", instructions: { task: "How well does this option match what the person asked for?", option: { kind: card.kind, title: card.title, when: card.when, venue: card.venue, price: card.priceText } }, criteria: LEVELS };
}

/** Named shows go first; everything else is scored by Jev in one call. Jev failure → date order. */
export async function rankCards(input: { text: string; filters: AskFilters; cards: AskCard[]; evaluate?: Evaluate }): Promise<{ ranked: AskCard[]; inputTokens: number; degraded: boolean }> {
  const named = input.cards.filter((card) => isNamed(card, input.filters.names)).map((card) => ({ ...card, score: 1 })).sort(byDate);
  const rest = input.cards.filter((card) => !isNamed(card, input.filters.names));
  if (rest.length === 0) return { ranked: named, inputTokens: 0, degraded: false };
  try {
    const questions = Object.fromEntries(rest.map((card, index) => [`c${index}`, question(card)]));
    const { answers, inputTokens } = await (input.evaluate ?? jevEvaluate)({ state: { request: input.text, filters: { ...input.filters } }, questions });
    const scored = rest.map((card, index) => {
      const answer = answers[`c${index}`];
      const score = answer?.type === "score" ? Math.min(1, Math.max(0, answer.score / (LEVELS.length - 1))) : 0;
      return { ...card, score };
    }).sort((left, right) => right.score - left.score || byDate(left, right));
    return { ranked: [...named, ...scored], inputTokens, degraded: false };
  } catch (error) {
    logAskFailure("jev", error);
    return { ranked: [...named, ...[...rest].sort(byDate)], inputTokens: 0, degraded: true };
  }
}
```

```ts
  return {
    textStream: result.textStream,
    usage: Promise.resolve(result.usage).then((usage) => ({ inputTokens: usage.inputTokens ?? 0, outputTokens: usage.outputTokens ?? 0 })),
  };
};

function system(language: "en" | "zh"): string {
  return language === "zh"
    ? "你是 KiwiCue 的奥克兰活动向导。用一句简短、口语化的中文回答，点名最推荐的那个选项并说明原因。不要编造列表之外的信息，不要用 Markdown。"
    : "You are KiwiCue's Auckland outing guide. Answer in one short, spoken-style English sentence naming the best option and why. Use only the listed options. No Markdown.";
}

/** Streams the one-sentence reply. Any failure just ends the stream; the cards already answer the question. */
export async function* replyDeltas(input: { text: string; language: "en" | "zh"; top: AskCard[]; stream?: StreamReply; timeoutMs?: number }): AsyncGenerator<string, TokenUsage> {
  if (input.top.length === 0) return NO_USAGE;
  const options = input.top.slice(0, 3).map(({ title, when, venue, priceText, kind }) => ({ kind, title, when, venue, price: priceText }));
  try {
    const result = (input.stream ?? deepseekStreamReply)({
      system: system(input.language), prompt: JSON.stringify({ question: input.text, options }),
      abortSignal: AbortSignal.timeout(input.timeoutMs ?? DEFAULT_TIMEOUT_MS),
    });
    for await (const delta of result.textStream) yield delta;
    return await result.usage;
  } catch (error) {
    logAskFailure("reply", error);
    return NO_USAGE;
  }
}
```

```ts
async function readRequest(request: Request): Promise<{ ok: true; data: AskRequest } | { ok: false; status: number }> {
  const raw = await request.text();
  if (new TextEncoder().encode(raw).length > MAX_BODY_BYTES) return { ok: false, status: 413 };
  try {
    const parsed = askRequestSchema.safeParse(JSON.parse(raw));
    return parsed.success ? { ok: true, data: parsed.data } : { ok: false, status: 400 };
  } catch (error) {
    if (error instanceof SyntaxError) return { ok: false, status: 400 };
    throw error;
  }

function stream(input: AskRequest, decision: { usageId: string; remaining: number; degrade: boolean; tier: AskTier }, deps: AskDeps): Response {
  const body = new ReadableStream<Uint8Array>({
    async start(controller) {
      const emit: Emit = (event) => controller.enqueue(encodeEvent(event));
      let usage = ZERO_USAGE;
      try {
        usage = await runAsk(input, { degrade: decision.degrade, now: deps.now(), deps }, emit);
        emit({ type: "done", remaining: decision.remaining, tier: decision.tier });
      } catch (error) {
        logAskFailure("stream", error);
        emit({ type: "error", code: "internal", message: messages[input.language].internal });
      } finally {
        await deps.quota?.recordCost(decision.usageId, costMicroUsd(usage), decision.degrade);
        controller.close();
      }
    },
  });
  return new Response(body, { headers: NDJSON });
}

export async function handleAsk(request: Request, deps: AskDeps): Promise<Response> {
  const read = await readRequest(request);
  if (!read.ok) return errorResponse("invalid", read.status, "en");
  const { data } = read;
  if (!deps.configured || !deps.quota) return errorResponse("unavailable", 503, data.language);
  const identity = await deps.identify(request);
  const decision = await deps.quota.reserve(identity.subject, identity.tier, identity.userId);
  if (!decision.allowed) return errorResponse(decision.reason, decision.reason === "unavailable" ? 503 : 429, data.language);
  return stream(data, decision, deps);
}
```

```ts
export type AskViewState = {
  status: "idle" | "thinking" | "streaming" | "done" | "error";
  text: string; filters: AskFilters | null; chips: AskChip[]; cards: AskCard[]; bestId: string | null;
  reply: string; notices: string[]; error: { code: AskErrorCode; message: string } | null; remaining: number | null; history: string[];

export const INITIAL_ASK_STATE: AskViewState = {
  status: "idle", text: "", filters: null, chips: [], cards: [], bestId: null, reply: "", notices: [], error: null, remaining: null, history: [],
};
type Action = { type: "start"; text: string } | { type: "event"; event: AskStreamEvent } | { type: "fail"; message: string } | { type: "reset" };

function applyEvent(state: AskViewState, event: AskStreamEvent): AskViewState {
  switch (event.type) {
    case "understood": return { ...state, status: "streaming", filters: event.filters, chips: event.chips };
    case "result": return { ...state, cards: [...state.cards, event.card] };
    case "best": return { ...state, bestId: event.id };
    case "reply": return { ...state, reply: state.reply + event.delta };
    case "notice": return { ...state, notices: [...new Set([...state.notices, event.message])] };
    case "done": return { ...state, status: "done", remaining: event.remaining, history: [...state.history, state.text].slice(-MAX_HISTORY) };
    case "error": return { ...state, status: "error", error: { code: event.code, message: event.message } };
  }
}

export function askReducer(state: AskViewState, action: Action): AskViewState {
  if (action.type === "reset") return INITIAL_ASK_STATE;
  if (action.type === "fail") return { ...state, status: "error", error: { code: "internal", message: action.message } };
  if (action.type === "start") return { ...state, status: "thinking", text: action.text, cards: [], bestId: null, reply: "", notices: [], error: null };
  return applyEvent(state, action.event);
}
```

```ts
export function useAskStream(fetchImpl: typeof fetch = fetch) {
  const [state, dispatch] = useReducer(askReducer, INITIAL_ASK_STATE);
  const latest = useRef(state);
  latest.current = state;
  const controller = useRef<AbortController | null>(null);

  const send = useCallback(async (text: string, language: Language, previous: AskFilters | null, refineOnly: boolean) => {
    controller.current?.abort();
    const abort = new AbortController();
    controller.current = abort;
    const history = latest.current.history;
    dispatch({ type: "start", text });
    try {
      const response = await fetchImpl("/api/ask", {
        method: "POST", signal: abort.signal, headers: { "content-type": "application/json" },
        body: JSON.stringify({ text, language, previous, history, refineOnly }),
      });
      if (!response.body) throw new TypeError("Empty response");
      for await (const event of readEvents(response.body)) dispatch({ type: "event", event });
    } catch (error) {
      if (error instanceof DOMException && error.name === "AbortError") return; // Superseded by a newer question.
      dispatch({ type: "fail", message: NETWORK_MESSAGE[language] });
    }
  }, [fetchImpl]);

  const ask = useCallback((text: string, language: Language) => send(text, language, latest.current.filters, false), [send]);
  const removeChip = useCallback((key: string, language: Language) => {
    const { filters, text } = latest.current;
    return filters ? send(text, language, clearChip(filters, key), true) : Promise.resolve();
  }, [send]);
  const reset = useCallback(() => { controller.current?.abort(); dispatch({ type: "reset" }); }, []);
  return { state, ask, removeChip, reset };
}
```

```ts
"use client";

import { useCallback, useEffect, useRef, useState, useSyncExternalStore } from "react";

type SpeechResultList = ArrayLike<ArrayLike<{ transcript: string }> & { isFinal: boolean }>;
type Recognition = {
  lang: string; interimResults: boolean; continuous: boolean;
  onresult: ((event: { resultIndex: number; results: SpeechResultList }) => void) | null;
  onerror: ((event: { error: string }) => void) | null;
  onend: (() => void) | null;
  start(): void; stop(): void; abort(): void;
};
export type SpeechRecognitionConstructor = new () => Recognition;

// The Web Speech API is not in TypeScript's DOM lib and ships prefixed in Chrome and Safari.
export function speechRecognitionConstructor(scope: object = typeof window === "undefined" ? {} : window): SpeechRecognitionConstructor | null {
  const candidate = (scope as Record<string, unknown>).SpeechRecognition ?? (scope as Record<string, unknown>).webkitSpeechRecognition;
  return typeof candidate === "function" ? candidate as SpeechRecognitionConstructor : null;
}

const noop = () => () => undefined;

export function useSpeechInput({ language, onFinal }: { language: "en" | "zh"; onFinal(text: string): void }) {
  const supported = useSyncExternalStore(noop, () => speechRecognitionConstructor() !== null, () => false);
  const [listening, setListening] = useState(false);
  const [interim, setInterim] = useState("");
  const [error, setError] = useState<"denied" | "failed" | null>(null);
  const recognition = useRef<Recognition | null>(null);
  const finalHandler = useRef(onFinal);
  useEffect(() => { finalHandler.current = onFinal; }, [onFinal]);
  useEffect(() => () => recognition.current?.abort(), []);

  const start = useCallback(() => {
    const Constructor = speechRecognitionConstructor();
    if (!Constructor) return;
    const instance = new Constructor();
    instance.lang = language === "zh" ? "zh-CN" : "en-NZ";
    instance.interimResults = true;
    instance.continuous = false;
    instance.onresult = (event) => {
      const latest = event.results[event.results.length - 1];
      const transcript = latest[0]?.transcript.trim() ?? "";
      if (latest.isFinal) { setInterim(""); if (transcript) finalHandler.current(transcript); } else setInterim(transcript);
    };
    instance.onerror = (event) => setError(event.error === "not-allowed" || event.error === "service-not-allowed" ? "denied" : "failed");
    instance.onend = () => setListening(false);
    recognition.current = instance;
    setError(null);
    setListening(true);
    instance.start();
  }, [language]);

  const stop = useCallback(() => recognition.current?.stop(), []);
  return { supported, listening, interim, error, start, stop };
}
```

```tsx
function readSpeakPreference(): boolean {
  try { return localStorage.getItem(SPEAK_KEY) === "1"; } catch (error) { return error instanceof DOMException ? false : false; }
}

function writeSpeakPreference(value: boolean): void {
  try { localStorage.setItem(SPEAK_KEY, value ? "1" : "0"); } catch (error) { if (!(error instanceof DOMException)) throw error; }
}

function useSpokenReply(reply: string, done: boolean, enabled: boolean, language: "en" | "zh") {
  useEffect(() => {
    if (!done || !enabled || !reply || typeof speechSynthesis === "undefined") return;
    const utterance = new SpeechSynthesisUtterance(reply);
    utterance.lang = language === "zh" ? "zh-CN" : "en-NZ";
    speechSynthesis.cancel();
    speechSynthesis.speak(utterance);
    return () => speechSynthesis.cancel();
  }, [done, enabled, language, reply]);
}

function trapFocus(event: KeyboardEvent<HTMLDivElement>, onClose: () => void) {
  if (event.key === "Escape") { onClose(); return; }
  if (event.key !== "Tab") return;
  const focusable = [...event.currentTarget.querySelectorAll<HTMLElement>("a[href], button:not([disabled]), input, [tabindex='0']")];
  if (focusable.length === 0) return;
  const [first, last] = [focusable[0], focusable[focusable.length - 1]];
  if (event.shiftKey && document.activeElement === first) { event.preventDefault(); last.focus(); }
  else if (!event.shiftKey && document.activeElement === last) { event.preventDefault(); first.focus(); }
}
```
